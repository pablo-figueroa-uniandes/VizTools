% \lpsection{number}{level}{title}: level 0 plain, 1 starred group, 2 part
\newcommand{\lpsection}[3]{%
  \ifnum#2=2
    \clearpage
    \phantomsection\addcontentsline{toc}{section}{#3}%
    {\noindent\LARGE\bfseries #3\par}\vspace{1.2em}%
  \fi
  \par\addvspace{\ifnum#2=0 0.9em\else 1.6em\fi}%
  \ifnum#2=1 \phantomsection\addcontentsline{toc}{subsection}{#1.\enspace #3}\fi
  \noindent\hypertarget{lpsec.#1}{}\textbf{#1.\ifnum#2=1 \enspace #3.\fi}\enspace\ignorespaces}

  \hyphenpenalty=10000\exhyphenpenalty=10000\frenchspacing
  \noindent #1\ \ensuremath{\mathrel{#2}\equiv}\par\nopagebreak}%
  {\par\endgroup\addvspace{0.2em}}
